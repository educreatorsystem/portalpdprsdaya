Kod Backend Google Apps Script (`Code.gs`)

Berikut adalah kod **Google Apps Script** yang telah disesuaikan dengan ID Sheet 16kLuWnLZECEtTKD50XTbXVAHIKcgwHvzesC-vuF3DV0 dan ID Folder Drive 13YwZnpAOfPNUKVO04Gdu4Tt1Q7Dedszh anda. Tampalkan kod ini ke dalam editor AppsScript anda untuk menyimpan data dan gambar secara berpusat. 

/* ====================================================================
 * SISTEM TUGASAN PDPR SK TAMAN DAYA - BACKEND GOOGLE APPS SCRIPT
 * SHEET ID : 16kLuWnLZECEtTKD50XTbXVAHIKcgwHvzesC-vuF3DV0
 * DRIVE ID : 13YwZnpAOfPNUKVO04Gdu4Tt1Q7Dedszh
 * ==================================================================== */

const SPREADSHEET_ID = '16kLuWnLZECEtTKD50XTbXVAHIKcgwHvzesC-vuF3DV0';
const FOLDER_DRIVE_ID = '13YwZnpAOfPNUKVO04Gdu4Tt1Q7Dedszh';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    if (action === 'save_tugasan') {
      let sheet = ss.getSheetByName('TUGASAN');
      if (!sheet) {
        sheet = ss.insertSheet('TUGASAN');
        sheet.appendRow(['ID', 'Tarikh', 'Nama Guru', 'Subjek', 'Kelas', 'Tajuk', 'T1 Ayat', 'T1 Video', 'T1 Kuiz', 'T2 Ayat', 'T2 Video', 'T2 Kuiz', 'Masa Dicipta']);
      }
      
      const id = data.id || 'TUG-' + Date.now();
      
      if (data.isEdit) {
        // Edit Tugasan
        const rows = sheet.getDataRange().getValues();
        for (let i = 1; i < rows.length; i++) {
          if (rows[i][0] == id) {
            sheet.getRange(i + 1, 2, 1, 11).setValues([[
              data.tarikh, data.namaGuru, data.subjek, data.kelas, data.tajuk,
              data.t1Ayat, data.t1Video, data.t1Kuiz, data.t2Ayat, data.t2Video, data.t2Kuiz
            ]]);
            break;
          }
        }
      } else {
        // Tugasan Baru
        sheet.appendRow([
          id, data.tarikh, data.namaGuru, data.subjek, data.kelas, data.tajuk,
          data.t1Ayat, data.t1Video, data.t1Kuiz, data.t2Ayat, data.t2Video, data.t2Kuiz,
          new Date()
        ]);
      }

      return ContentService.createTextOutput(JSON.stringify({ status: 'success', id: id }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'delete_tugasan') {
      let sheet = ss.getSheetByName('TUGASAN');
      if (sheet) {
        const rows = sheet.getDataRange().getValues();
        for (let i = 1; i < rows.length; i++) {
          if (rows[i][0] == data.id) {
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'deleted' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'submit_murid') {
      let sheet = ss.getSheetByName('PENGHANTARAN_MURID');
      if (!sheet) {
        sheet = ss.insertSheet('PENGHANTARAN_MURID');
        sheet.appendRow(['ID', 'Masa', 'Tarikh Tugasan', 'Nama Murid', 'Kelas', 'Subjek', 'Tajuk', 'Status', 'URL Gambar Drive']);
      }

      let imageUrl = '';
      if (data.imageBase64) {
        try {
          const folder = DriveApp.getFolderById(FOLDER_DRIVE_ID);
          const contentType = data.imageBase64.substring(5, data.imageBase64.indexOf(';'));
          const bytes = Utilities.base64Decode(data.imageBase64.substr(data.imageBase64.indexOf('base64,') + 7));
          const blob = Utilities.newBlob(bytes, contentType, 'BUKTI_' + data.namaMurid.replace(/\s+/g, '_') + '_' + Date.now() + '.jpg');
          const file = folder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          imageUrl = file.getUrl();
        } catch (err) {
          imageUrl = 'Ralat Muat Naik Gambar: ' + err.message;
        }
      }

      sheet.appendRow([
        'SUB-' + Date.now(),
        new Date(),
        data.tarikh,
        data.namaMurid,
        data.kelas,
        data.subjek,
        data.tajuk,
        'SELESAI',
        imageUrl
      ]);

      return ContentService.createTextOutput(JSON.stringify({ status: 'success', imageUrl: imageUrl }))
        .setMimeType(ContentService.MimeType.JSON);
    }

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Sistem PDPR SK Taman Daya Backend Active.");
}
